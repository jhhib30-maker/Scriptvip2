local function applyCustomFace(head, faceId)
	if not head or not faceId then return end
	local tex = "rbxassetid://" .. tostring(faceId)
	local function ensureFace()
		local ex = head:FindFirstChild("custom_face")
		if ex and ex.Texture == tex then return end
		for _, d in ipairs(head:GetChildren()) do
			if d:IsA("Decal") then d:Destroy() end
		end
		local dec = Instance.new("Decal")
		dec.Name = "custom_face"; dec.Texture = tex
		dec.Face = Enum.NormalId.Front; dec.Transparency = 0; dec.Parent = head
	end
	ensureFace()
	head.ChildAdded:Connect(function(c)
		if c:IsA("Decal") and c.Name:lower() == "face" then c:Destroy(); task.defer(ensureFace) end
	end)
	head.ChildRemoved:Connect(function(c)
		if c:IsA("Decal") then task.defer(ensureFace) end
	end)
end

local function changeToEpicHead(char)
	local oldHead = char:FindFirstChild("Head")

	if not oldHead or not hum then return end

	for _, acc in ipairs(char:GetChildren()) do

			local handle = acc:FindFirstChild("Handle")

				if weld.Part0 == oldHead or weld.Part1 == oldHead then
					table.insert(headAccessories, acc:Clone())
				end
			end

	end
	local desc = Instance.new("HumanoidDescription")

		return Players:CreateHumanoidModelFromDescriptionAsync(desc, Enum.HumanoidRigType.R15)

local function httpGetJson(url)
	local httpRequest = request or http_request or (syn and syn.request) or (http and http.request)
	if httpRequest then
		local ok, response = pcall(function()
			return httpRequest({ Url = url, Method = "GET", Headers = { ["User-Agent"] = "Mozilla/5.0" } })
		end)
		if ok and response and response.Body then
			local decOk, decoded = pcall(function() return HttpService:JSONDecode(response.Body) end)
			if decOk and type(decoded) == "table" then return true, decoded end
		end
	end
	local okGet, body = pcall(function() return game:HttpGet(url) end)
	if okGet and body then
		local decOk, decoded = pcall(function() return HttpService:JSONDecode(body) end)
		if decOk and type(decoded) == "table" then return true, decoded end
	end
	return false, nil
end

local function parseColor(s)
	if typeof(s) == "Color3" then return s end
	if type(s) == "string" then
		local r, g, b = s:match("([%d.]+),%s*([%d.]+),%s*([%d.]+)")
		if r then return Color3.new(tonumber(r), tonumber(g), tonumber(b)) end
	end
	return Color3.new(1, 1, 1)
end

local function getAnimId(id)
	local ok, obj = pcall(function() return game:GetObjects("rbxassetid://" .. math.floor(id)) end)
	if ok and #obj > 0 then
		local r = obj[1]
		if r:IsA("Animation") and r.AnimationId ~= "" then return r.AnimationId end
		for _, d in ipairs(r:GetDescendants()) do
			if d:IsA("Animation") and d.AnimationId ~= "" then return d.AnimationId end
		end
	end
	return "rbxassetid://" .. tostring(id)
end

local function prepClone(clone)
	for _, d in ipairs(clone:GetDescendants()) do
		if d:IsA("Motor6D") or d:IsA("Weld") or d:IsA("WeldConstraint") then d:Destroy()
		elseif d:IsA("Decal") and d.Name:lower() == "face" then d:Destroy() end
	end
	clone.Anchored = false; clone.CanCollide = false; clone.Massless = true
end

local function attachAccessory(character, acc)
	local handle = acc:FindFirstChild("Handle"); if not handle then return end
	local handleAtt = handle:FindFirstChildOfClass("Attachment"); if not handleAtt then return end
	local target
	for _, d in ipairs(character:GetDescendants()) do
		if d:IsA("Attachment") and d.Name == handleAtt.Name and d.Parent:IsA("BasePart") then target = d; break end
	end
	if not target then return end
	local base = target.Parent; local c = handle:Clone(); prepClone(c)
	local mesh = c:FindFirstChildOfClass("SpecialMesh"); if mesh then mesh.Scale = Vector3.new(1, 1, 1) end
	c.CFrame = base.CFrame * target.CFrame * handleAtt.CFrame:Inverse(); c.Parent = character
	local w = Instance.new("Weld"); w.Part0 = base; w.Part1 = c; w.C0 = target.CFrame; w.C1 = handleAtt.CFrame; w.Parent = c
end

local moodState = {

local function getAnimator()
	local char = player.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not hum then return nil end
	return hum:FindFirstChildOfClass("Animator") or hum:WaitForChild("Animator", 3)

local function stopMood()
	if moodState.currentTrack then
		pcall(function() moodState.currentTrack:Stop(0) end)
		pcall(function() moodState.currentTrack:Destroy() end)
		moodState.currentTrack = nil
	end
	moodState.currentId = nil; moodState.currentName = nil; moodState.currentFaceId = nil
end

local function playMood(id, name, faceId)
	local animator = getAnimator(); if not animator then return end
	stopMood()
	local animId = getAnimId(id); local a = Instance.new("Animation"); a.AnimationId = animId
	local t = animator:LoadAnimation(a)
	t.Priority = moodState.cfg.priority; t.Looped = moodState.cfg.looped
	t:AdjustWeight(moodState.cfg.weight); t:AdjustSpeed(moodState.cfg.speed)
	moodState.currentTrack = t; moodState.currentId = id; moodState.currentName = name; moodState.currentFaceId = faceId
	t:Play()
	if faceId then
		local char = player.Character; local head = char:FindFirstChild("Head")
		for _, p in ipairs(char:GetChildren()) do if p.Name == "Head" and p.Transparency == 0 then head = p end end
		applyCustomFace(head, faceId)
	end
	if not moodState.cfg.looped then
		t.Stopped:Connect(function() if moodState.currentTrack == t and moodState.currentId == id then t:Play() end end)
	end
	log("Play mood: " .. (name or tostring(id)))
end

local function refreshMood()
	if not moodState.currentTrack then return end
	moodState.currentTrack.Priority = moodState.cfg.priority
	moodState.currentTrack.Looped = moodState.cfg.looped
	moodState.currentTrack:AdjustWeight(moodState.cfg.weight)
	moodState.currentTrack:AdjustSpeed(moodState.cfg.speed)
end

local currentEmoteTrack = nil

local function playEmote(assetId, name)

	if currentEmoteTrack then currentEmoteTrack:Stop(); currentEmoteTrack = nil end

	track.Priority = Enum.AnimationPriority.Action4; track.Looped = false; track:Play()

	log("💃 Đang chạy emote: " .. name)
end

					if assetType == 48 then animIds.Climb = item.id
					elseif assetType == 49 then animIds.Death = item.id
					elseif assetType == 50 then animIds.Fall = item.id
					elseif assetType == 51 then animIds.Idle = item.id
					elseif assetType == 52 then animIds.Jump = item.id
					elseif assetType == 53 then animIds.Run = item.id
					elseif assetType == 54 then animIds.Swim = item.id
					elseif assetType == 55 then animIds.Walk = item.id

		for _, track in pairs(hum:GetPlayingAnimationTracks()) do
			track:Stop()
		end
		local animator = hum:FindFirstChild("Animator") or Instance.new("Animator", hum)

			if not animObject then return end

						anim.AnimationId = hashId
						return anim
					end
				end
				model:Destroy()
				return nil
			end)

		end

		local runAnim = bypassWithKeyframeSequence(animIds.Run)
		local walkAnim = bypassWithKeyframeSequence(animIds.Walk)

		if idleAnim or runAnim or walkAnim then

		local function getRealAnimation(id)
			if not id then return nil end
			local ok, objs = pcall(function() return game:GetObjects("rbxassetid://" .. id) end)
			if ok and objs and #objs > 0 then
				for _, obj in ipairs(objs) do
					if obj:IsA("Animation") then return obj end
				end
			end
			return nil
		end

		local idleReal = getRealAnimation(animIds.Idle)
		local runReal = getRealAnimation(animIds.Run)
		local walkReal = getRealAnimation(animIds.Walk)

		if idleReal then playAnim(idleReal) end

		if idleReal or runReal or walkReal then
			log("✅ Đã áp animation (Bypass GetObjects): " .. name)

local BASE64_CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

local function toBase64(str)
	local out = {}
	for i = 1, #str, 3 do
		local b1 = str:byte(i) or 0
		local b2 = str:byte(i + 1)
		local b3 = str:byte(i + 2)

		local n = b1 * 65536 + (b2 or 0) * 256 + (b3 or 0)
		local c1 = math.floor(n / 262144) % 64 + 1
		local c2 = math.floor(n / 4096) % 64 + 1
		local c3 = b2 and (math.floor(n / 64) % 64 + 1) or nil
		local c4 = b3 and (n % 64 + 1) or nil

		table.insert(out, BASE64_CHARS:sub(c1, c1))
		table.insert(out, BASE64_CHARS:sub(c2, c2))
		table.insert(out, c3 and BASE64_CHARS:sub(c3, c3) or "=")
		table.insert(out, c4 and BASE64_CHARS:sub(c4, c4) or "=")
	end
	return "AVH1:" .. table.concat(out)
end

local function fromBase64(encoded)
	if type(encoded) ~= "string" then return nil end
	local clean = encoded:gsub("^%s+", ""):gsub("%s+", "")
	if clean:sub(1, 5) == "AVH1:" then
		clean = clean:sub(6)
	end
	if #clean == 0 or #clean % 4 ~= 0 then return nil end

	local map = {}
	for i = 1, #BASE64_CHARS do
		map[BASE64_CHARS:sub(i, i)] = i - 1
	end

	local out = {}
	for i = 1, #clean, 4 do
		local ch1 = clean:sub(i, i)
		local ch2 = clean:sub(i + 1, i + 1)
		local ch3 = clean:sub(i + 2, i + 2)
		local ch4 = clean:sub(i + 3, i + 3)

		local v1 = map[ch1]
		local v2 = map[ch2]
		if not v1 or not v2 then return nil end

		local v3 = (ch3 == "=") and 0 or map[ch3]
		local v4 = (ch4 == "=") and 0 or map[ch4]
		if not v3 or not v4 then return nil end

		local n = v1 * 262144 + v2 * 4096 + v3 * 64 + v4
		local b1 = math.floor(n / 65536) % 256
		local b2 = math.floor(n / 256) % 256
		local b3 = n % 256

		table.insert(out, string.char(b1))
		if ch3 ~= "=" then table.insert(out, string.char(b2)) end
		if ch4 ~= "=" then table.insert(out, string.char(b3)) end
	end
	return table.concat(out)
end

local function buildAvatarPayload(customName)
	local character = player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	if not humanoid then return nil end

	local desc = humanoid:GetAppliedDescription()
	local saveData = {
		name = (customName and customName ~= "") and customName or ("Avatar_" .. os.time()),
		thumbnail = getThumbnail(player.UserId),
		bodyColors = {
			headColor = tostring(desc.HeadColor), torsoColor = tostring(desc.TorsoColor),
			leftArmColor = tostring(desc.LeftArmColor), rightArmColor = tostring(desc.RightArmColor),
			leftLegColor = tostring(desc.LeftLegColor), rightLegColor = tostring(desc.RightLegColor),
		},
		bodyTypeScale = desc.BodyTypeScale or 1.0, headAssetId = desc.Head,
		assets = {}, animations = {}, headInfo = {}, headDecals = {}, mood = nil,
	}

	local head = character:FindFirstChild("Head")
	if head then
		saveData.headInfo = {
			name = head.Name, size = {head.Size.X, head.Size.Y, head.Size.Z},
			color = tostring(head.Color), transparency = head.Transparency,
			material = tostring(head.Material), className = head.ClassName,
		}
		local mesh = head:FindFirstChildOfClass("SpecialMesh")
		if mesh then
			saveData.headInfo.mesh = {
				meshId = mesh.MeshId, textureId = mesh.TextureId,
				scale = {mesh.Scale.X, mesh.Scale.Y, mesh.Scale.Z},
				offset = {mesh.Offset.X, mesh.Offset.Y, mesh.Offset.Z},
			}
		end
		if head:IsA("MeshPart") then saveData.headInfo.meshId = head.MeshId; saveData.headInfo.textureId = head.TextureID end
		for _, d in ipairs(head:GetChildren()) do
			if d:IsA("Decal") then table.insert(saveData.headDecals, {name = d.Name, texture = d.Texture, face = tostring(d.Face), transparency = d.Transparency}) end
		end
	end

	local okAcc, accList = pcall(function() return desc:GetAccessories(true) end)
	if okAcc and accList then
		for _, acc in ipairs(accList) do
			if acc.AssetId and acc.AssetId ~= 0 then table.insert(saveData.assets, {type = "Accessory", id = acc.AssetId}) end
		end
	end

	for _, v in ipairs(character:GetChildren()) do
		if v:IsA("Shirt") then table.insert(saveData.assets, {type = "Shirt", template = v.ShirtTemplate})
		elseif v:IsA("Pants") then table.insert(saveData.assets, {type = "Pants", template = v.PantsTemplate}) end
	end

	local animate = character:FindFirstChild("Animate")
	if animate then
		local anims = {}
		for _, d in ipairs(animate:GetDescendants()) do
			if d:IsA("Animation") and d.AnimationId ~= "" then table.insert(anims, {name = d.Name, id = d.AnimationId}) end
		end
		saveData.animations = anims
	end

	if moodState.currentId then saveData.mood = {id = moodState.currentId, name = moodState.currentName, faceId = moodState.currentFaceId} end
	return saveData
end

local function saveAvatar(customName)
	local saveData = buildAvatarPayload(customName)
	if not saveData then return false end
	local all = readJSON(FILE_NAME); table.insert(all, saveData); writeJSON(FILE_NAME, all)
	log("Đã lưu: " .. saveData.name); return true
end

local function importAvatarCode(shareCode, customName)
	local rawJson = fromBase64(shareCode)
	if not rawJson then
		local trimmed = tostring(shareCode or ""):gsub("^%s+", ""):gsub("%s+$", "")
		if trimmed:sub(1, 1) == "{" then
			rawJson = trimmed
		end
	end
	if not rawJson then return nil, "Mã chia sẻ không hợp lệ" end

	local okDec, decoded = pcall(function() return HttpService:JSONDecode(rawJson) end)
	if not okDec or type(decoded) ~= "table" then
		return nil, "Không đọc được dữ liệu avatar"
	end

	decoded.name = (customName and customName ~= "") and customName or ((decoded.name or "Imported") .. " (Share)")
	local all = readJSON(FILE_NAME)
	table.insert(all, decoded)
	writeJSON(FILE_NAME, all)
	return decoded, nil
end

local function loadAvatar(data)
	local character = player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	if not humanoid or not data then return false end

	if _G.PreviewImage then _G.PreviewImage.Image = data.thumbnail or getThumbnail(player.UserId) or "" end

	local bc = character:FindFirstChildOfClass("BodyColors")
	if bc and data.bodyColors then
		bc.HeadColor3 = parseColor(data.bodyColors.headColor); bc.TorsoColor3 = parseColor(data.bodyColors.torsoColor)
		bc.LeftArmColor3 = parseColor(data.bodyColors.leftArmColor); bc.RightArmColor3 = parseColor(data.bodyColors.rightArmColor)
		bc.LeftLegColor3 = parseColor(data.bodyColors.leftLegColor); bc.RightLegColor3 = parseColor(data.bodyColors.rightLegColor)
	end

	for _, v in ipairs(character:GetChildren()) do
		if v:IsA("Accessory") or v:IsA("Shirt") or v:IsA("Pants") then v:Destroy() end
	end

	for _, item in ipairs(data.assets or {}) do
		if item.type == "Shirt" and item.template then local s = Instance.new("Shirt"); s.ShirtTemplate = item.template; s.Parent = character
		elseif item.type == "Pants" and item.template then local p = Instance.new("Pants"); p.PantsTemplate = item.template; p.Parent = character end
	end

	for _, item in ipairs(data.assets or {}) do
		if item.type == "Accessory" and item.id then
			local okGet, objs = pcall(function() return game:GetObjects("rbxassetid://" .. item.id) end)
			if okGet and objs and #objs > 0 then
				local src = objs[1]
				if src:IsA("Accessory") then attachAccessory(character, src)
				elseif src:IsA("Model") then for _, c in ipairs(src:GetChildren()) do if c:IsA("Accessory") then attachAccessory(character, c) end end end
			end
		end
	end

	if data.headAssetId and data.headAssetId ~= 0 then
		local cur = humanoid:GetAppliedDescription()
		if cur.Head ~= data.headAssetId then cur.Head = data.headAssetId; pcall(function() humanoid:ApplyDescription(cur) end); task.wait(0.15) end
	end

	local head = character:FindFirstChild("Head")
	if head then
		head.Transparency = 0
		if data.headInfo and data.headInfo.mesh then
			local mesh = head:FindFirstChildOfClass("SpecialMesh") or Instance.new("SpecialMesh", head)
			mesh.MeshId = data.headInfo.mesh.meshId or ""; mesh.TextureId = data.headInfo.mesh.textureId or ""
			if data.headInfo.mesh.scale then mesh.Scale = Vector3.new(data.headInfo.mesh.scale[1], data.headInfo.mesh.scale[2], data.headInfo.mesh.scale[3]) end
		end
		for _, v in ipairs(head:GetChildren()) do if v:IsA("Decal") then v:Destroy() end end
		if data.headDecals then
			for _, dec in ipairs(data.headDecals) do
				local d = Instance.new("Decal"); d.Name = dec.name or "face"; d.Texture = dec.texture
				d.Face = Enum.NormalId[dec.face] or Enum.NormalId.Front; d.Transparency = dec.transparency or 0; d.Parent = head
			end
		end
	end

	if data.animations then
		local animate = character:FindFirstChild("Animate")
		if animate then
			for _, a in ipairs(data.animations) do
				local obj = animate:FindFirstChild(a.name, true)
				if obj and obj:IsA("Animation") then obj.AnimationId = a.id end
			end
			animate.Disabled = true; task.wait(0.1); animate.Disabled = false
		end
	end

	if data.mood and data.mood.id then
		task.spawn(function() task.wait(0.2); playMood(data.mood.id, data.mood.name, data.mood.faceId); if _G.SyncMoodUI then _G.SyncMoodUI(data.mood.id) end end)
	else stopMood() end
	log("Load xong: " .. (data.name or "?")); return true
end

local function makeSlider(parent, label, y, min, max, def, cb)
	local t = Instance.new("TextLabel")
	t.Size = UDim2.new(1, -16, 0, 14); t.Position = UDim2.new(0, 8, 0, y)
	t.BackgroundTransparency = 1; t.Text = label; t.TextColor3 = Theme.Sub
	t.Font = Enum.Font.GothamSemibold; t.TextSize = 11; t.TextXAlignment = Enum.TextXAlignment.Left; t.Parent = parent
	local bg = Instance.new("TextButton")
	bg.Size = UDim2.new(1, -16, 0, 12); bg.Position = UDim2.new(0, 8, 0, y + 16)
	bg.BackgroundColor3 = Theme.Item; bg.BorderSizePixel = 0; bg.Text = ""; bg.AutoButtonColor = false; bg.Parent = parent
	Instance.new("UICorner", bg).CornerRadius = UDim.new(0, 6)
	local fill = Instance.new("Frame")
	fill.Size = UDim2.new((def - min) / (max - min), 0, 1, 0); fill.BackgroundColor3 = Theme.Accent1
	fill.BorderSizePixel = 0; fill.Parent = bg; Instance.new("UICorner", fill).CornerRadius = UDim.new(0, 6)
	local val = Instance.new("TextLabel")
	val.Size = UDim2.new(1, -16, 0, 14); val.Position = UDim2.new(0, 8, 0, y + 30)
	val.BackgroundTransparency = 1; val.Text = string.format("%.2f", def); val.TextColor3 = Theme.Accent1
	val.Font = Enum.Font.GothamBold; val.TextSize = 12; val.Parent = parent
	local dragging = false
	local function setFromX(x)
		local s = math.clamp((x - bg.AbsolutePosition.X) / bg.AbsoluteSize.X, 0, 1)
		local v = math.floor((min + s * (max - min)) * 100) / 100
		fill.Size = UDim2.new(s, 0, 1, 0); val.Text = string.format("%.2f", v); cb(v)
	end
	bg.InputBegan:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = true; setFromX(i.Position.X) end end)
	UserInputService.InputChanged:Connect(function(i) if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then setFromX(i.Position.X) end end)
	UserInputService.InputEnded:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = false end end)
end

local function runRawScript(codeStr, titleStr)
	if not codeStr or codeStr == "" then
		log("❌ Script rỗng, không thể chạy")
		return
	end
	local fn, compErr = loadstring(codeStr)
	if not fn then
		log("❌ Lỗi cú pháp script (" .. titleStr .. "): " .. tostring(compErr))
		return
	end
	log("▶ Đang chạy script: " .. titleStr)
	task.spawn(function()
		local ok, runErr = pcall(fn)
		if not ok then
			log("❌ Lỗi khi thực thi (" .. titleStr .. "): " .. tostring(runErr))
		else
			log("✅ Đã chạy xong: " .. titleStr)
		end
	end)
end

local function searchScriptBlox(keyword)
	clearScriptResults()
	local q = (keyword or ""):gsub("^%s+", ""):gsub("%s+$", "")
	if q == "" then
		scriptStatusLbl.Text = "Nhập từ khóa"
		return
	end

	scriptStatusLbl.Text = "⏳ Đang tìm..."
	log("🔍 ScriptBlox tìm kiếm: " .. q)

	local url = "https://scriptblox.com/api/script/search?q=" .. HttpService:UrlEncode(q) .. "&page=1&max=20"

	task.spawn(function()
		local ok, decoded = httpGetJson(url)
		if not ok or not decoded then
			scriptStatusLbl.Text = "Lỗi API"
			log("❌ Không lấy được dữ liệu từ ScriptBlox API")
			return
		end

		local scripts = (decoded.result and decoded.result.scripts) or decoded.scripts or {}
		if #scripts == 0 then
			scriptStatusLbl.Text = "0 kết quả"
			log("⚠️ Không tìm thấy script nào cho: " .. q)
			return
		end

		for _, item in ipairs(scripts) do
			addScriptCard(item)
		end

		scriptStatusLbl.Text = #scripts .. " kết quả"
		log("✅ Tìm thấy " .. #scripts .. " script cho: " .. q)
	end)
end

local function searchMarketEmotes(keyword)
	clearEmoteResults(); if keyword == "" then return end
	log("🔍 Đang tìm emote: " .. keyword)
	local url = "https://catalog.roblox.com/v1/search/items/details?Category=12&Subcategory=39&Keyword="
		.. HttpService:UrlEncode(keyword) .. "&Limit=30"
	task.spawn(function()
		local httpRequest = request or http_request or (syn and syn.request) or (http and http.request)
		if not httpRequest then log("❌ Executor không hỗ trợ HTTP request!"); return end
		local ok, response = pcall(function() return httpRequest({ Url = url, Method = "GET" }) end)
		if not (ok and response and response.Body) then log("❌ Lỗi mạng"); return end
		local decodeOk, data = pcall(function() return HttpService:JSONDecode(response.Body) end)
		if not (decodeOk and data and data.data) then log("❌ Dữ liệu không hợp lệ"); return end
		local count = 0
		for _, item in ipairs(data.data) do
			if item.id and (item.assetType == 61 or item.assetType == nil) then
				addEmoteCard({ id = item.id, name = item.name or ("Emote " .. item.id) }); count += 1
			end
		end
		log("✅ Đã thêm " .. count .. " emote")
		task.defer(function() emoteScroll.CanvasSize = UDim2.new(0, 0, 0, emoteGrid.AbsoluteContentSize.Y + 20) end)
	end)

local function searchAnimations(keyword)
	clearAnimResults()
	if keyword == "" then return end
	log("🔍 Đang tìm animation bundle: " .. keyword)

	local url = "https://catalog.roblox.com/v1/search/items/details?Category=12&Subcategory=38&Keyword="
		.. HttpService:UrlEncode(keyword) .. "&Limit=30"

	task.spawn(function()
		local httpRequest = request or http_request or (syn and syn.request) or (http and http.request)
		if not httpRequest then log("❌ Executor không hỗ trợ HTTP request!"); return end
		local ok, response = pcall(function() return httpRequest({ Url = url, Method = "GET" }) end)
		if not (ok and response and response.Body) then log("❌ Lỗi mạng"); return end
		local decodeOk, data = pcall(function() return HttpService:JSONDecode(response.Body) end)
		if not (decodeOk and data and data.data) then log("❌ Dữ liệu không hợp lệ"); return end

		local count = 0
		for _, item in ipairs(data.data) do
			if item.id and item.name then
				addAnimCard({ id = item.id, name = item.name, assetType = item.assetType })
				count += 1
			end
		end
		log("✅ Đã thêm " .. count .. " animation bundle")
		task.defer(function() animScroll.CanvasSize = UDim2.new(0, 0, 0, animGrid.AbsoluteContentSize.Y + 20) end)
	end)
end
